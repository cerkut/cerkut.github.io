document.addEventListener("DOMContentLoaded", function () {
  const input = document.getElementById("thesis-search");
  const list = document.getElementById("theses-list");
  if (!input || !list) return;

  const years = Array.prototype.slice.call(list.querySelectorAll(":scope > h2.bibliography"));

  const filter = (term) => {
    term = term.trim().toLowerCase();
    years.forEach((h2) => {
      let anyVisible = false;
      let node = h2.nextElementSibling;
      while (node && !(node.tagName === "H2" && node.classList.contains("bibliography"))) {
        if (node.tagName === "H4") {
          const title = node;
          const detail = node.nextElementSibling; // matching blockquote
          const text = (title.textContent + " " + (detail ? detail.textContent : "")).toLowerCase();
          const match = term === "" || text.indexOf(term) !== -1;
          title.style.display = match ? "" : "none";
          if (detail) detail.style.display = match ? "" : "none";
          if (match) anyVisible = true;
          node = detail ? detail.nextElementSibling : node.nextElementSibling;
          continue;
        }
        node = node.nextElementSibling;
      }
      h2.style.display = anyVisible ? "" : "none";
    });
  };

  let timeoutId;
  input.addEventListener("input", function () {
    clearTimeout(timeoutId);
    const value = this.value;
    timeoutId = setTimeout(() => filter(value), 150);
  });
});
</script>
